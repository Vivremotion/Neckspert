`CHROMATIC_NOTES` commence par C : l'indice d'une note est aussi sa classe de hauteur (`midi % 12`).

```ts
import {
	CHROMATIC_NOTES,
	MAX_PLAYABLE_FRET,
	STANDARD_TUNING,
	pitchOfMidi,
	type Midi,
	type PitchName,
	type StringNumber
} from '$lib/domain/music/types';
import type { LocateTarget } from '$lib/domain/types';

const OCTAVE = 12;

function assertFret(fret: number): void {
	if (!Number.isInteger(fret) || fret < 0 || fret > MAX_PLAYABLE_FRET) {
		throw new RangeError(`Fret must be an integer from 0 to ${MAX_PLAYABLE_FRET}: ${fret}`);
	}
}

export function midiOnString(string: StringNumber, fret: number): Midi {
	assertFret(fret);
	return STANDARD_TUNING[string] + fret;
}

export function pitchOnString(string: StringNumber, fret: number): PitchName {
	return pitchOfMidi(midiOnString(string, fret));
}

/** Frets 0..maxFret on `string` whose pitch is `pitch`, ascending. */
export function fretsForPitch(
	string: StringNumber,
	pitch: PitchName,
	maxFret: number = MAX_PLAYABLE_FRET
): number[] {
	assertFret(maxFret);
	const openPitchClass = STANDARD_TUNING[string] % OCTAVE;
	const lowestFret = (CHROMATIC_NOTES.indexOf(pitch) - openPitchClass + OCTAVE) % OCTAVE;
	const frets: number[] = [];
	for (let fret = lowestFret; fret <= maxFret; fret += OCTAVE) frets.push(fret);
	return frets;
}

/** MIDI notes accepted for a mode 1 target: every octave of the pitch on that string, frets 0..MAX_PLAYABLE_FRET. */
export function acceptableMidis(target: LocateTarget): Midi[] {
	return fretsForPitch(target.string, target.pitch).map((fret) =>
		midiOnString(target.string, fret)
	);
}
```

```ts
import type { NoteLabel, PitchName } from '$lib/domain/music/types';

const FLAT_NAMES: Partial<Record<PitchName, string>> = {
	'C#': 'Db',
	'D#': 'Eb',
	'F#': 'Gb',
	'G#': 'Ab',
	'A#': 'Bb'
};

/** Naturals keep their name; altered pitches get sharp or flat at random (random() < 0.5 → sharp). Naturals are labelled 'sharp' and draw nothing. */
export function spellPitch(pitch: PitchName, random: () => number = Math.random): NoteLabel {
	if (!FLAT_NAMES[pitch]) return { pitch, spelling: 'sharp' };
	return { pitch, spelling: random() < 0.5 ? 'sharp' : 'flat' };
}

export function formatNote(label: NoteLabel): string {
	return label.spelling === 'flat' ? (FLAT_NAMES[label.pitch] ?? label.pitch) : label.pitch;
}

export function answerChoiceLabel(pitch: PitchName): string {
	const flat = FLAT_NAMES[pitch];
	return flat ? `${pitch} / ${flat}` : pitch;
}
```

- Produces: `classifyNameConfusion(expected: PitchName, answered: PitchName): Confusion | null`.

```ts
import type { Confusion } from '$lib/domain/types';

/** `semitones` is never 0: equal pitch classes are handled by the callers. */

	return Math.abs(semitones) <= SEMITONE_CONFUSION_MAX
		? { kind: 'semitone', delta: semitones as SemitoneDelta }

	const to = CHROMATIC_NOTES.indexOf(answered);
	const up = (to - from + OCTAVE) % OCTAVE;
	return fromDistance(up >= OCTAVE / 2 ? up - OCTAVE : up);
}
```

```ts
import { CHROMATIC_NOTES, pitchOfMidi, type Midi, type PitchName } from '$lib/domain/music/types';
import type { Confusion, LocateTarget } from '$lib/domain/types';
import { acceptableMidis } from './fretboard';

const OCTAVE = 12;

/** Farthest distance, in semitones, still counted as a near miss. */
const SEMITONE_CONFUSION_MAX = 2;

type SemitoneDelta = -2 | -1 | 1 | 2;

/** `semitones` is never 0: equal pitch classes are handled by the callers. */
function fromDistance(semitones: number): Confusion {
	return Math.abs(semitones) <= SEMITONE_CONFUSION_MAX
		? { kind: 'semitone', delta: semitones as SemitoneDelta }
		: { kind: 'other', semitones };
}

/** Null when the played note is acceptable. */
export function classifyNoteConfusion(target: LocateTarget, playedMidi: Midi): Confusion | null {
	const acceptable = acceptableMidis(target);
	if (acceptable.includes(playedMidi)) return null;
	if (pitchOfMidi(playedMidi) === target.pitch) return { kind: 'octave' };
	// `acceptable` is ascending and the comparison strict, so a tie keeps the lower note.
	const nearest = acceptable.reduce((best, midi) =>
		Math.abs(playedMidi - midi) < Math.abs(playedMidi - best) ? midi : best
	);
	return fromDistance(playedMidi - nearest);
}

/** Null when the names match. Delta is the signed pitch-class distance folded into −6..+5. */
export function classifyNameConfusion(expected: PitchName, answered: PitchName): Confusion | null {
	if (expected === answered) return null;
	const from = CHROMATIC_NOTES.indexOf(expected);
	const to = CHROMATIC_NOTES.indexOf(answered);
	const up = (to - from + OCTAVE) % OCTAVE;
	return fromDistance(up >= OCTAVE / 2 ? up - OCTAVE : up);
}
```

```ts
import { MAX_PROMPT_FRET, type StringNumber } from '$lib/domain/music/types';
import type { Curriculum, CurriculumSet, IdentifyTarget, LocateTarget } from '$lib/domain/types';
import { pitchOnString } from './fretboard';

/** Teaching order: low E string first. */
const CURRICULUM_STRINGS: readonly StringNumber[] = [6, 5, 4, 3, 2, 1];

/** Set `a` of each string: the pitches under the fret markers 3, 5 and 7. */
const SET_A_FRETS: readonly number[] = [3, 5, 7];

const PROMPT_FRETS: readonly number[] = Array.from(
	{ length: MAX_PROMPT_FRET + 1 },
	(_, fret) => fret
);

const SET_PARTS = ['a', 'b'] as const;

const SET_FRETS: Record<(typeof SET_PARTS)[number], readonly number[]> = {
	a: SET_A_FRETS,
	b: PROMPT_FRETS.filter((fret) => !SET_A_FRETS.includes(fret))
};

function locateTarget(string: StringNumber, fret: number): LocateTarget {
	return { direction: 'locate', string, pitch: pitchOnString(string, fret) };
}

function identifyTarget(string: StringNumber, fret: number): IdentifyTarget {
	return { direction: 'identify', string, fret };
}

export function buildCurriculum(): Curriculum {
	const sets: CurriculumSet[] = [];
	let previousLocateId: string | null = null;
	for (const string of CURRICULUM_STRINGS) {
		for (const part of SET_PARTS) {
			const frets = SET_FRETS[part];
			const locateId = `locate-s${string}-${part}`;
			sets.push({
				id: locateId,
				direction: 'locate',
				targets: frets.map((fret) => locateTarget(string, fret)),
				requires: previousLocateId
			});
			sets.push({
				id: `identify-s${string}-${part}`,
				direction: 'identify',
				targets: frets.map((fret) => identifyTarget(string, fret)),
				requires: locateId
			});
			previousLocateId = locateId;
		}
	}
	return sets;
}

/** Every mode 1 target: strings 6 to 1, frets 0 to 11 on each (72 targets). */
export function placementTargets(): LocateTarget[] {
	return CURRICULUM_STRINGS.flatMap((string) =>
		PROMPT_FRETS.map((fret) => locateTarget(string, fret))
	);
}
```
